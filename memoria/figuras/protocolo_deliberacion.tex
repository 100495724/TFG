	\begin{tikzpicture}[
		font = \footnotesize,
		ag/.style     = {rectangle, rounded corners=2pt, draw, thick,

		ciego/.style  = {ag, very thick, fill=black!8},
		estado/.style = {rectangle, draw, dashed, minimum width=88mm,
			minimum height=6mm, align=center, fill=black!3},
		fl/.style     = {-{Stealth[length=1.8mm]}}
		]
		
		% Génesis
		\node[ciego] (g1) at (0,0)   {\texttt{agent\_1} (ciego)};
		\node[ag]    (g2) at (3.2,0) {\texttt{agent\_2}};
		\node[ag]    (g3) at (6.4,0) {\texttt{agent\_3}};
		\node[left=3mm of g1, align=right] {$t=0$\\génesis};
		
		\node[estado] (e0) at (3.2,-1.3) {estado $t=0$};
		\draw[fl] (g1) -- (e0);
		\draw[fl] (g2) -- (e0);
		\draw[fl] (g3) -- (e0);
		
		% Turno 1
		\node[ciego] (a1) at (0,-2.6)   {\texttt{agent\_1} (ciego)};
		\node[ag]    (a2) at (3.2,-2.6) {\texttt{agent\_2}};
		\node[ag]    (a3) at (6.4,-2.6) {\texttt{agent\_3}};
		\node[left=3mm of a1, align=right] {$t=1$};
		
		\draw[fl] (e0) -- (a1);
		\draw[fl] (e0) -- (a2);
		\draw[fl] (e0) -- (a3);
		
		\node[estado] (e1) at (3.2,-3.9) {estado $t=1$};
		\draw[fl] (a1) -- (e1);
		\draw[fl] (a2) -- (e1);
		\draw[fl] (a3) -- (e1);
		
		\node at (3.2,-4.6) {$\vdots$};
		\node at (3.2,-5.2) {$t=2,3,4$};
		
	\end{tikzpicture}
	\caption[Protocolo de deliberación]{Secuencia de interacción del comité. En
		génesis los tres agentes deciden de forma independiente. En cada turno
		posterior, los tres reciben el estado del turno anterior y sus respuestas
		no lo sustituyen hasta que los tres han respondido, de modo que la
		actualización es síncrona.}
